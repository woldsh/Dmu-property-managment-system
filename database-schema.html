2. Firestore Security Rules

In Firebase NoSQL, this file (firestore.rules) acts as your schema definition and access control matrix, replacing SQL CREATE TABLE statements.

firestore.rules 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // USERS COLLECTION
    match /users/{userId} {
      // Anyone logged in can read users directories
      allow read: if request.auth != null;
      // Users can edit themselves, MDs can edit anyone
      allow write: if request.auth != null && (
        request.auth.uid == userId || 
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.userRole == 'managing_director'
      );
    }

    // MATERIALS INVENTORY COLLECTION
    match /materials/{materialId} {
      allow read: if request.auth != null;
      // Only system admins and stock clerks can inject/modify base materials
      allow write: if request.auth != null && (
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.userRole in [
          'managing_director', 
          'fixed_asset_stock_clerk', 
          'consumable_item_stock_clerk'
        ]
      );
    }

    // MATERIAL REQUESTS (Aggregate Workflow)
    match /Request_materials/{requestId} {
      allow read: if request.auth != null;
      // Any authenticated user can spawn a request
      allow create: if request.auth != null;
      // Update cascades handled by approvers
      allow update: if request.auth != null;
    }

    // USER REPORT (Actual Material Withdrawal Ledger)
    match /User-Report/{reportId} {
      // Requesters can see their own, admins/keepers see all
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

    // PICKUP VERIFICATIONS
    match /Send_to_Users/{sendId} {
      // Created by Store Keepers, read by Requesters
      allow read, write: if request.auth != null;
    }

    // PEER-TO-PEER TRANSFERS
    match /Material_transfers/{transferId} {
      allow read, write: if request.auth != null;
    }

    // INVENTORY AUDITS
    match /cycle_counts/{countId} {
      allow read, write: if request.auth != null;
    }
  }
}

3. Initialization Script (Seed)

To programmatically create your database structure and inject initial data (like MySQL INSERT INTO), run a Node.js script using the Firebase Admin SDK.

seed_database.js 
const admin = require('firebase-admin');

// 1. Initialize Firebase Admin
const serviceAccount = require('./serviceAccountKey.json');
admin.initializeApp({
  credential: admin.credential.cert(serviceAccount)
});

const db = admin.firestore();

async function initializeDatabase() {
  console.log("🚀 Bootstrapping Firestore Database Architecture...");

  try {
    // 2. Inject Primary Administrator Account
    const adminRef = db.collection('users').doc('SYS_ADMIN_001');
    await adminRef.set({
      uid: 'SYS_ADMIN_001',
      displayName: 'Master Administrator',
      email: 'admin@pms-system.local',
      userRole: 'managing_director',
      mainRole: 'admin',
      department: 'Executive',
      status: 'active',
      createdAt: admin.firestore.FieldValue.serverTimestamp()
    });
    console.log("✅ Collection [users] initialized with admin root node.");

    // 3. Inject Initial Category Base Material
    const materialRef = db.collection('materials').doc('MAT_TEST_001');
    await materialRef.set({
      materialName: 'Latitude 7420 Developer Machine',
      materialCode: 'IT-DEV-LT-001',
      quantity: 50,
      condition: 'New',
      category: 'IT Hardware',
      unit: 'Units',
      materialType: 'fixed_asset',
      description: 'Standard issue developer laptop.',
      storeLocation: 'Zone A-1',
      createdAt: admin.firestore.FieldValue.serverTimestamp()
    });
    console.log("✅ Collection [materials] initialized with base inventory.");

    console.log("🎉 Database structure successfully seeded!");
    // Note: Request_materials, User-Report, etc., will auto-generate 
    // dynamically when your application code executes addDoc().
    
    process.exit(0);
  } catch (error) {
    console.error("❌ Seeding failed: ", error);
    process.exit(1);
  }
}

initializeDatabase();
